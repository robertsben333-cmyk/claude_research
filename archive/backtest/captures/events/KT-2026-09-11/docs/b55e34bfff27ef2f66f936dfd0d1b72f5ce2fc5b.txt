	// 검색년도
	function loadYear() {
		// api 데이터
		var data = {};
		data.langDivType = 'ENG';	// 언어 구분 타입
		// api 데이터

		kt_common.api({
			'url' : '/corp/announcements/v1.0/achievements/years',
			'type' : 'get',
			'data' : data,
			'callback' : callbackYear
		});
	}

	// 검색년도 콜백함수

		try {
			if(type === 'success') {

					let yearList = result.data.yeareList;

					if(yearList.length > 0) {
						var optionsHtml = '';
						$j.each(yearList, function(idx, data) {
							if (idx > 0) {
								if (data ' + data + ' ';
						});
						$j('#earnings_year').append(optionsHtml); // 년도세팅

	// 검색결과
	function search() {
		// 재생영역 초기화
		if(isOnair === false || $j('.play_record_on_air').is(':visible') === false) {
			clear();
		}

		// api 데이터
		var data = {};
		data.langDivType = 'ENG';	// 언어 구분 타입
		/*
		var year = $j('#earnings_year').val();
		if(year !== '') {
			data.yyInfo = year;
		}*/
		// api 데이터

		kt_common.api({
			'url' : '/corp/announcements/v1.0/achievements',
			'type' : 'get',
			'data' : data,
			'callback' : callbackSearch
		});
	}

	// 검색결과 콜백함수
	function callbackSearch(type, result) {
		let year = $j('#earnings_year').val();
		try {
			if(type === 'success') {
				if(kt_common.isNull(result.data) !== '') {
					resultList = result.data.achPreList;
					viewList = $j.grep(resultList, function(element, index) { return element.yyInfo == year});

					if(viewList.length > 0) {
						var html = '';
						$j.each(viewList, function(idx, data) {
							var pdf = kt_common.isNull(data.saveFilePath01) + kt_common.isNull(data.saveFileNm01);
							var excel = kt_common.isNull(data.saveFilePath02) + kt_common.isNull(data.saveFileNm02);
							var greetingSound = kt_common.isNull(data.saveFilePath03) + kt_common.isNull(data.saveFileNm03);
							var qnaSound = kt_common.isNull(data.saveFilePath05) + kt_common.isNull(data.saveFileNm05);
							var rowTitle = parseInt(data.qtrType) + 'Q '+ data.yyInfo+' ';

							html += ' ';
							html += '	 ' + parseInt(data.qtrType) + 'Q ';
							html += '	 ';
							html += '		 ';

							if(pdf !== '') {
								html += '			  PDF download     ';
							}

							if(excel !== '') {
								html += '			  Excel download     ';
							}

							html += '		 ';
							html += '	 ';
							html += '	 ';

							//if(greetingSound !== '') {  add 20171222 onair 추가 : 우측 화면 표시를 위해 인사말은 항상 보임.
								html += '		  Listening  ';
							//}

							html += '	 ';
							html += '	 ';

							if(qnaSound !== '') {
								html += '		  Listening  ';
							}

							html += '	 ';
							html += ' ';
						});

						$j('#earnings_list').find('tbody').html(html);

						var $btnSound = $j('#earnings_list').find('.btn-listen');
						$btnSound.on('click', playSound);

						if(isOnair === false || $j('.play_record_on_air').is(':visible') === false) {	// 온에어가 아닐경우
							if($btnSound.length > 0) {
								playSound();
							}
						}
					}
				}
			}
		} catch(e) {
			common_log.log('응답파싱오류가 발생하였습니다.(' + e.message + ')');
		}
	}

	// 음성재생
	function playSound(event) {
		var repositoryIdx = 0;
		var soundIdx = 0;
		var $target;
		if(event === undefined) {
			$target = $j('#earnings_list').find('.btn-listen').eq(0);
		} else {
			$target = $j(event.target);
			if(event.target.tagName !== 'BUTTON') {
				$target = $target.parent();
			}
		}

		if($target.hasClass('active') === true) {
			return false;
		}

		$j('#earnings_list').find('.btn-listen').removeClass('active');
		$target.addClass('active');
		$j('#earnings_list').find('tr').removeClass('active');
		$target.parent().parent().addClass('active');

		repositoryIdx = $target.parent().parent().index();
		soundIdx = $target.parent().index() - 2;

		var playData = viewList[repositoryIdx];

		// title, 일시, 발표자
		$j('#earnings_title').text(parseInt(playData.qtrType) + 'Q ' + $j('#earnings_year').val() + ' Conference Call');
		$j('#earnings_date').text(playData.ptDateInfo);
		$j('#earnings_present').text(playData.ptUser);

		// 플레이어세팅
		var sound;
		var caption;
		var text;
		if(soundIdx === 1) {
			sound = kt_common.isNull(playData.saveFilePath05) + kt_common.isNull(playData.saveFileNm05);
			caption = kt_common.isNull(playData.saveFilePath06) + kt_common.isNull(playData.saveFileNm06);
			text = kt_common.isNull(playData.vcSubtitText05);
		} else {
			sound = kt_common.isNull(playData.saveFilePath03) + kt_common.isNull(playData.saveFileNm03);
			caption = kt_common.isNull(playData.saveFilePath04) + kt_common.isNull(playData.saveFileNm04);
			text = kt_common.isNull(playData.vcSubtitText03);
		}

		var captionHtml = ' Subtitle for voice file ';
		if(caption !== '') {
			captionHtml += '  Subtitle download    ';
		}
		captionHtml += ' ' + text + ' ';
		$j('.play-record').find('.box-subtitles').html(captionHtml);

		$j('#earnings_player').attr('src', '/eng' + sound);
		// $j('#earnings_player')[0].play();

		//add 20171222 onair 추가
		if(kt_common.isNull(playData.aedPtType) == '01'){
			// 발표 전,
			showOnAir(repositoryIdx);
		}else{
			// 발표 후,
			$j('.play-record').show();
			$j('.playWarp').show();
			$j('.play_record_on_air').hide();

			isOnair = false;
		}
	}

	// add 20171222 onair 추가
	function showOnAir(idx) {
		var onAirData = viewList[idx];

		// 이미지 정보
		var banrLinkType = kt_common.isNull(onAirData.banrLinkType);
		var banrLinkUrl = kt_common.isNull(onAirData.banrLinkUrl);
		var banrSaveImgNm = kt_common.isNull(onAirData.banrSaveImgNm);
		var banrSaveImgPath = kt_common.isNull(onAirData.banrSaveImgPath);
		var banrAltText = kt_common.isNull(onAirData.banrAltText);
		var aTitleText = " Earnings Release Broadcast Listening";
		if(banrLinkType == "01"){
			// 새창
			$j('.play_record_on_air').find("a").attr("target",  "_blank");
			aTitleText = "[new window]" + aTitleText;
		}else{
			// 현재창
			$j('.play_record_on_air').find("a").removeAttr("target");
			aTitleText = "[current window]" + aTitleText;
		}
		$j('.play_record_on_air').find("a").attr("href",  banrLinkUrl);
		$j('.play_record_on_air').find("a").attr("title",  aTitleText);
		$j('.play_record_on_air').find("img").attr("src", '/eng' + banrSaveImgPath+banrSaveImgNm)
		$j('.play_record_on_air').find("img").attr("alt", banrAltText);

		$j('.play_record_on_air').show();
		$j('.play-record').show();
		$j('.playWarp').hide();

		isOnair = true;
	}

	// 음성재성영역 초기화
	function clear() {
		$j('.play-record').hide();
		$j('#earnings_title').text('');
		$j('#earnings_date').text('');
		$j('#earnings_present').text('');
		$j('.play-record').find('.box-subtitles').html('');
		$j('#earnings_player').attr('src', '');
	}
