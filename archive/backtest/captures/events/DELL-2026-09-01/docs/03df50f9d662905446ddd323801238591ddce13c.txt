          function gate() { return document.querySelector("[data-range-gate]"); }

          function focusables(dialog) {
            return Array.prototype.filter.call(
              dialog.querySelectorAll("a[href], button:not([disabled])"),
              function (el) { return el.offsetParent !== null; }
            );
          }

          function open(trigger) {
            var g = gate();
            if (!g) { return false; }
            // Safari does not focus a button on click, so activeElement is not
            // a reliable fallback on its own — the caller passes the button.
            lastTrigger = trigger || (document.activeElement !== document.body ? document.activeElement : null);
            g.hidden = false;
            var close = g.querySelector(".tv-rangegate__close");
            if (close) { close.focus(); }
            return true;
          }

          function close() {
            var g = gate();
            if (!g || g.hidden) { return; }
            g.hidden = true;
            if (lastTrigger && typeof lastTrigger.focus === "function") { lastTrigger.focus(); }
            lastTrigger = null;
          }

          document.addEventListener("click", function (e) {
            if (e.target.closest && e.target.closest("[data-range-gate-close]")) { close(); }
          });

          document.addEventListener("keydown", function (e) {
            var g = gate();
            if (!g || g.hidden) { return; }
            if (e.key === "Escape") { close(); return; }
            if (e.key !== "Tab") { return; }

            var dialog = g.querySelector(".tv-rangegate__dialog");
            if (!dialog) { return; }
            var items = focusables(dialog);
            if (!items.length) { return; }

            var first = items[0];
            var last = items[items.length - 1];
            if (!dialog.contains(document.activeElement)) {
              e.preventDefault();
              first.focus();
            } else if (e.shiftKey && document.activeElement === first) {
              e.preventDefault();
              last.focus();
            } else if (!e.shiftKey && document.activeElement === last) {
              e.preventDefault();
              first.focus();
            }
          });

          window.AltIndexRangeGate = { open: open, close: close };
        })();

        function debounce(func, wait, immediate) {
            var timeout;
            return function() {
                var context = this, args = arguments;
                var later = function() {
                    timeout = null;
                    if (!immediate) func.apply(context, args);
                };
                var callNow = immediate && !timeout;
                clearTimeout(timeout);
                timeout = setTimeout(later, wait);
                if (callNow) func.apply(context, args);
            };
        }
