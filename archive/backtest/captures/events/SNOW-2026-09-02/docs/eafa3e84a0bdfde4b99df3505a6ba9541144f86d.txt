      /* Billboard parallax: keep the reveal gap equal to the ad's real height.
         GAM sets the creative size, so measure it and expose --billboard-h;
         ResizeObserver catches the creative rendering and any viewport change. */
      (function () {
        var ad = document.getElementById('PYMNT_ad');
        if (!ad) return;
        // Floor matching billboard-ad.css's own reserved min-heights (100/90/250,
        // same breakpoints). A no-fill/ad-blocked impression flips #PYMNT_ad to
        // .pymnts-ad--empty, which forces min-height:0 !important -- collapsing
        // the ad's real rect well below what was reserved at first paint. Before
        // this mattered (main.css/legacy-2022-baseline.css were fully
        // render-blocking), that collapse always resolved before first paint, so
        // it was never visible. Now that those can be deferred on this bucket,
        // an empty-ad collapse mid-page-load is a real, measurable CLS source --
        // confirmed via the Layout Instability API (a real capture isolating this
        // from every other Stage 6 change: identical shift regardless of
        // main.css/baseline's own load speed, present only when the billboard ad
        // resolves empty). Never letting the gap shrink below the reservation
        // means a no-fill impression keeps its reserved whitespace instead of
        // visibly collapsing -- an accepted trade (unused space on the pages
        // that don't fill) for eliminating that CLS on the pages that don't.
        function reservedFloor() {
          var w = window.innerWidth;
          if (w >= 1280) return 250;
          if (w >= 768) return 90;
          return 100;
        }
        function sync() {
          // Align the first page sheet flush with the pinned ad's bottom edge.
          // Deriving it from the ad's real rect (minus any html offset the admin
          // bar adds) works whether the admin bar pushes the page down or just
          // overlays it. getBoundingClientRect() is scroll-independent for a
          // fixed element, so the value stays correct while scrolling.
          var htmlMarginTop = parseFloat(getComputedStyle(document.documentElement).marginTop) || 0;
          var gap = Math.max(ad.getBoundingClientRect().bottom - htmlMarginTop, reservedFloor());
          document.documentElement.style.setProperty('--billboard-h', gap + 'px');
        }
        sync();
        if (window.ResizeObserver) {
          new ResizeObserver(sync).observe(ad);
        }
        window.addEventListener('resize', sync);
        // billboard-ad.css is loaded async (preload + onload swap), so the ad
        // becomes fixed/offset AFTER this inline script first runs. ResizeObserver
        // only catches size changes, not the position change, so re-measure once
        // the stylesheet has applied: on load, on the next frame, and a short
        // backstop timeout.
        window.addEventListener('load', function () {
          sync();
          requestAnimationFrame(sync);
        });
        requestAnimationFrame(sync);
        setTimeout(sync, 300);
      })();

	 * Mobile navigation drawer open/close (replaces the old Bootstrap collapse).

	(function () {
		const toggle = document.getElementById('mobile-toggle');
		const drawer = document.getElementById('mobile-drawer');
		if (!toggle || !drawer) return;

		let lockedScrollY = 0;
		const header = document.querySelector('header.main-nav');

		// The header isn't sticky/pinned — it scrolls away like any other
		// content. --drawer-top (see 2022/navigation-menus.css) had assumed it
		// always sits 68px (+ any billboard ad) tall at the top of the viewport,
		// which is only true at scrollY 0; scrolled any further, that reserved
		// space showed whatever page content was actually at the top instead.
		// Measuring the header's real remaining visible height fixes that for
		// any scroll position, including 0.
		const syncDrawerTop = () => {
			if (!header) return;
			const visibleBottom = Math.max(0, header.getBoundingClientRect().bottom);
			document.documentElement.style.setProperty('--drawer-top', visibleBottom + 'px');
		};

		const setDrawer = (open) => {
			if (open) syncDrawerTop();
			toggle.setAttribute('aria-expanded', String(open));
			toggle.setAttribute('aria-label', open ? 'Close navigation' : 'Open navigation');
			drawer.classList.toggle('is-open', open);
			drawer.setAttribute('aria-hidden', String(!open));
			// aria-hidden alone doesn't stop keyboard focus from reaching the
			// drawer's contents while it's visually hidden — inert does.
			drawer.toggleAttribute('inert', !open);
			document.body.classList.toggle('drawer-open', open);
			// position:fixed (not overflow:hidden) locks background scroll without
			// losing the scroll position: overflow:hidden alone makes scrollY jump
			// to 0, which fires a real 'scroll' event — and this page's scroll-based
			// billboard-ad-hiding logic reads that as "scrolled back to top" and
			// re-shows the ad mid-drawer-open, with nothing re-syncing the drawer's
			// --billboard-h-based offset to match, opening a gap above the drawer.
			if (open) {
				lockedScrollY = window.scrollY;
				document.body.style.position = 'fixed';
				document.body.style.top = `-${lockedScrollY}px`;
				document.body.style.width = '100%';
			} else {
				document.body.style.position = '';
				document.body.style.top = '';
				document.body.style.width = '';
				// This restore should be an imperceptible snap back (the page never
				// really moved — it only looked scrolled via the negative top
				// offset above), not an animated scroll. :root has scroll-behavior:
				// smooth site-wide (2022-style.css), which window.scrollTo() would
				// otherwise inherit, animating the restore into a visible jump.
				window.scrollTo({ top: lockedScrollY, left: 0, behavior: 'instant' });
			}
			// preventScroll: true — when the toggle has scrolled out of view (the
			// header isn't sticky), focusing it would otherwise trigger the
			// browser's default scroll-into-view behavior, which — combined with
			// the site's global scroll-behavior: smooth — animates a visible
			// scroll on close, on top of the instant restore just above.
			if (!open) toggle.focus({ preventScroll: true });
		};

		toggle.addEventListener('click', () => setDrawer(toggle.getAttribute('aria-expanded') !== 'true'));

		document.addEventListener('keydown', (event) => {
			if (event.key === 'Escape' && toggle.getAttribute('aria-expanded') === 'true') setDrawer(false);
		});

		drawer.addEventListener('click', (event) => {
			if (event.target.closest('a')) setDrawer(false);
		});

		window.addEventListener('resize', () => {
			if (window.innerWidth > 991 && toggle.getAttribute('aria-expanded') === 'true') setDrawer(false);
		});
	})();
