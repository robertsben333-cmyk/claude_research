				createEvent: function(elem, action, fn) {
					if (elem.addEventListener) {
						elem.addEventListener(action, fn, false);
					} else if (elem.attachEvent) {
						elem['e' + action + fn] = fn;
						elem[action + fn] = function() { elem['e' + action + fn](window.event); };
						elem.attachEvent('on' + action, elem[action + fn]);
					} else {
						elem['on' + action] = elem['e' + action + fn];
					}
				}
