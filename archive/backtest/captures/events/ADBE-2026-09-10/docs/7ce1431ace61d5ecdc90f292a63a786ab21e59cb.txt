        // Navbar stock search function (redirects to analysis page)
        function navbarSearchStock(event) {
            if (event) event.preventDefault();

            // Check both desktop and mobile search inputs
            const desktopSearch = document.getElementById('navStockSearch');
            const mobileSearch = document.getElementById('navStockSearchMobile');
            
            let ticker = '';
            if (desktopSearch && desktopSearch.value) {
                ticker = desktopSearch.value.trim().toUpperCase();
            } else if (mobileSearch && mobileSearch.value) {
                ticker = mobileSearch.value.trim().toUpperCase();
            }

            if (ticker) {
                // Every nav search goes to the Investment Advisor deep link
                window.location.href = `/apps/investment-advisor/${ticker}/`;
            } else {
                alert('Please enter a stock ticker');
            }
        }

        // Stock search autocomplete (navbar + any page form via window.attachStockSuggest)
        (function() {
            function setup(inputId, boxId, onPick) {
                const input = document.getElementById(inputId);
                const box = document.getElementById(boxId);
                if (!input || !box) return;
                let timer = null, activeIdx = -1, items = [];

                function hide() { box.classList.remove('show'); box.innerHTML = ''; activeIdx = -1; items = []; }

                function go(ticker) {
                    input.value = ticker;
                    hide();
                    if (onPick) { onPick(ticker); return; }
                    window.location.href = '/apps/investment-advisor/' + ticker + '/';
                }

                function esc(s) {
                    return String(s).replace(/[&<>"']/g, function(c) {
                        return {'&':'&amp;',' ':'&gt;','"':'&quot;',"'":'&#39;'}[c];
                    });
                }

                function render(results) {
                    if (!results.length) { hide(); return; }
                    box.innerHTML = results.map(function(r) {
                        return ' ' +
                               ' ' + esc(r.ticker) + ' ' +
                               ' ' + esc(r.name || '') + '  ';
                    }).join('');
                    box.classList.add('show');
                    activeIdx = -1;
                    items = box.querySelectorAll('.nav-suggest-item');
                    items.forEach(function(el) {
                        el.addEventListener('mousedown', function(e) {
                            e.preventDefault();
                            go(el.dataset.ticker);
                        });
                    });
                }

                input.addEventListener('input', function() {
                    clearTimeout(timer);
                    const q = input.value.trim();
                    if (q.length  = 0) {
                        e.preventDefault();
                        go(items[activeIdx].dataset.ticker);
                    } else if (e.key === 'Escape') {
                        hide();
                    }
                });

                input.addEventListener('blur', function() { setTimeout(hide, 150); });
            }
            setup('navStockSearch', 'navSuggest');
            setup('navStockSearchMobile', 'navSuggestMobile');
            window.attachStockSuggest = setup;
            // Investment Advisor hero search form (present on advisor pages)
            setup('stockTicker', 'heroSuggest', function(t) {
                if (window.searchStock) { window.searchStock(t); }
                else { window.location.href = '/apps/investment-advisor/' + t + '/'; }
            });

        // Open modal
        function openModal() {
            feedbackModal.style.display = 'flex';
            feedbackSuccess.style.display = 'none';
            feedbackError.style.display = 'none';
        }
        if (feedbackBtn) {
            feedbackBtn.addEventListener('click', openModal);
        }

        // Open via #feedback hash (used by "contact support" links site-wide)
        if (window.location.hash === '#feedback') {
            openModal();
        }
        document.addEventListener('click', function(e) {
            var a = e.target.closest && e.target.closest('a[href="#feedback"], a[href$="#feedback"]');
            if (a) {
                e.preventDefault();
                openModal();
            }
        });

        // Close modal
        function closeModal() {
            feedbackModal.style.display = 'none';
            feedbackForm.reset();
        }

        if (feedbackModalClose) {
            feedbackModalClose.addEventListener('click', closeModal);
        }

        if (feedbackCancelBtn) {
            feedbackCancelBtn.addEventListener('click', closeModal);
        }

        // Close on outside click
        feedbackModal.addEventListener('click', function(e) {
            if (e.target === feedbackModal) {
                closeModal();
            }
        });

        // Form submission
        feedbackForm.addEventListener('submit', function(e) {
            e.preventDefault();

            var category = document.getElementById('feedbackCategory').value;
            var message = document.getElementById('feedbackMessage').value;
            var includeUrl = document.getElementById('feedbackIncludeUrl').checked;

            feedbackSubmitBtn.disabled = true;
            feedbackSubmitBtn.textContent = 'Sending...';

            fetch('/api/submit-feedback/', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-CSRFToken': getCookie('csrftoken')
                },
                body: JSON.stringify({
                    category: category,
                    message: message,
                    page_url: includeUrl ? window.location.href : '',
                    user_agent: navigator.userAgent
                })
            })
            .then(function(response) { return response.json(); })
            .then(function(data) {
                if (data.success) {
                    feedbackSuccess.style.display = 'block';
                    feedbackError.style.display = 'none';
                    feedbackForm.reset();
                    setTimeout(closeModal, 2000);
                } else {
                    feedbackError.style.display = 'block';
                    feedbackSuccess.style.display = 'none';
                }
            })
            .catch(function(error) {
                feedbackError.style.display = 'block';
                feedbackSuccess.style.display = 'none';
            })
            .finally(function() {
                feedbackSubmitBtn.disabled = false;
                feedbackSubmitBtn.textContent = 'Send Feedback';
            });
        });
