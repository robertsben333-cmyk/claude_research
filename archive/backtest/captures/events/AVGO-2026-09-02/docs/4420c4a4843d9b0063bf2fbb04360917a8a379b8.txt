function sortTable(table, columnIndex, ascending = true) {
  const tbody = table.querySelector('tbody');
  const rows = Array.from(tbody.querySelectorAll('tr'));
  const header = rows.shift();
  const headerth = Array.from(header.querySelectorAll('th'));
  headerth.forEach(thcell => thcell.style.backgroundColor = '#333333');
  headerth[columnIndex].style.backgroundColor='#864020';
  const headertxt = headerth[columnIndex].innerHTML;
  tbody.remove();

  rows.sort((rowA, rowB) => {
    let cellA = rowA.querySelectorAll('td')[columnIndex].textContent.trim().replace('$','').replaceAll(',','');
    let cellB = rowB.querySelectorAll('td')[columnIndex].textContent.trim().replace('$','').replaceAll(',','');
    const dataType = header.querySelectorAll('th')[columnIndex].dataset.type || 'string';

    let comparison = 0;
    if (dataType === 'number') {
        if (cellA.includes('M') || cellA.includes('B')) { cellA = cellA.replace('M','000000').replace('B','000000000').replace('.',''); }
        if (cellB.includes('M') || cellB.includes('B')) { cellB = cellB.replace('M','000000').replace('B','000000000').replace('.',''); }
        comparison = parseFloat(cellA) - parseFloat(cellB);
    } else if (dataType === 'date') {
        comparison = new Date(cellA) - new Date(cellB);
    } else {
        if (columnIndex == 0) {
        comparison = cellB.localeCompare(cellA);
        } else {
        comparison = cellA.localeCompare(cellB);
        }
    }
    return ascending ? comparison : -comparison;
  });
  rows.forEach(row => tbody.appendChild(row));
  table.appendChild(tbody);
  headerth[columnIndex].innerHTML = headertxt;
}
document.querySelectorAll('th').forEach((header, index) => {
  header.addEventListener('click', () => {
    const table = header.closest('table');
    const isAscending = header.classList.contains('asc');
    sortTable(table, index, !isAscending);
    header.classList.toggle('asc', !isAscending);
    header.classList.toggle('desc', isAscending);
    document.querySelectorAll('th').forEach(h => {
        if (h !== header) {
            h.classList.remove('asc', 'desc');
        }
    });
  });
});
