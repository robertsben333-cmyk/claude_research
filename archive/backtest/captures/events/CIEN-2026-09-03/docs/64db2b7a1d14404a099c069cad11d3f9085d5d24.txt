(function(){
  const channel = new BroadcastChannel('hf_email_login_channel');

  channel.onmessage = function(ev){
    const data = ev.data;
    if(!data || data.type !== 'email_login_success') return;

    // sanity check: if pending email exists in localStorage, match it
    const pending = localStorage.getItem('hf_pending_email');
    if(pending && data.payload && data.payload.email && pending.toLowerCase() !== data.payload.email.toLowerCase()){
      console.warn('Pending email mismatch. Ignoring broadcast.');
      return;
    }

    // close any login modals and call your existing login_user helper
    try { $.modal.close(); } catch(e) {}
    $.modal.close();
    
    $.modal('     Logging In  ', 
            {overlayClose: 0, escClose: 0, overlayId:"contact-overlay", 
                 containerId:"loginModal", containerCss:{  width:330}, position: ['15%']});
                 
    login_user_modal({
      email: data.payload.email,
      user_id: data.payload.user_id,
      user_level: data.payload.user_level,
      user_picture: data.payload.user_picture,
      first_name: data.payload.first_name,
      last_name: data.payload.last_name
    });

    // cleanup
    localStorage.removeItem('hf_pending_email');
    localStorage.removeItem('hf_pending_selector');
  };
})();
