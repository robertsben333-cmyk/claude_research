                  MarketBeat calculates consensus analyst ratings for stocks using the most recent rating from each Wall Street analyst that has rated a stock within the last twelve months. Each analyst's rating is normalized to a standardized rating score of 1 (sell), 2 (hold), 3 (buy) or 4 (strong buy). Analyst consensus ratings scores are calculated using the mean average of the number of normalized sell, hold, buy and strong buy ratings from Wall Street analysts. Each stock's consensus analyst rating is derived from its calculated consensus ratings score (0 to .5 = Strong Sell, .5 to 1 = Sell, 1 to 1.5 = Reduce, 1.5 to 2.5 = Hold, 2.5 to 3.0 = Moderate Buy, 3.0 to 3.5 = Buy, >3.5 = Strong Buy). MarketBeat's consensus price targets are a mean average of the most recent available price targets set by each analyst that has set a price target for the stock in the last twelve months.  MarketBeat's consensus ratings and consensus price targets may differ from those calculated by other firms due to differences in methodology and available data.

            var email = document.getElementById('txtLoginOnModalEmail').value;
            if (email == "") {
                notice.innerHTML = " Please enter your email address. ";
                return;
            }
            document.getElementById('btnMagicLinkOnModal').classList.add("active-loading");
            var returnUrl = window.location.pathname + window.location.search;
            console.log('[magiclink] sending', { tokenLength: (token || '').length, returnUrlLength: returnUrl.length });

            // POST, not GET: the reCAPTCHA token alone can blow past the 2048-char query string limit
            $.ajax({
                type: 'POST',
                url: '/scripts/modal/EmailChecker.ashx',
                data: {
                    command: 'magiclink',
                    email: email,
                    returnUrl: returnUrl,
                    recaptchaToken: token
                },
                success: function (data) {
                    var btn = document.getElementById('btnMagicLinkOnModal');
                    btn.classList.remove("active-loading");
                    console.log('[magiclink] response', data);
                    if (data.indexOf('c-red') > -1) {
                        notice.innerHTML = data;
                        // v2 invisible tokens are single-use â€” reset or the next click yields nothing
                        if (typeof grecaptcha !== 'undefined') { grecaptcha.reset(); }
                    } else {
                        notice.innerHTML = " " + data + " ";
                        btn.disabled = true;
                        btn.classList.remove('btn-secondary');
                        btn.classList.add('btn-disabled');
                    }
                },
                error: function (xhr, textStatus, errorThrown) {
                    document.getElementById('btnMagicLinkOnModal').classList.remove("active-loading");
                    console.error('[magiclink] failed', {
                        status: xhr.status,               // 404 = IIS rejected the request (404.15 = query string too long)
                        statusText: xhr.statusText,
                        textStatus: textStatus,
                        errorThrown: errorThrown,
                        tokenLength: (token || '').length,
                        response: (xhr.responseText || '').substring(0, 1000)
                    });
                    notice.innerHTML = " Something went wrong sending your login link. Please try again. ";
                    if (typeof grecaptcha !== 'undefined') { grecaptcha.reset(); }
                }
            });
        }

        // Toggle modal magic link button style based on email input
        document.addEventListener('input', function (e) {
            if (e.target.id === 'txtLoginOnModalEmail') {
                var btn = document.getElementById('btnMagicLinkOnModal');
                if (btn) {
                    var hasEmail = e.target.value.trim() !== '';
                    btn.classList.toggle('btn-disabled', !hasEmail);
                    btn.classList.toggle('btn-secondary', hasEmail);
                }
            }
        });
