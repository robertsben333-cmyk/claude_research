		function updateSlideshow() {
			// Move to next image
			currentIndex = (currentIndex + 1) % totalImages;
			var nextImage = slideshowArray[currentIndex];

			// Create a new image element for crossfade
			var $newImg = $(' ', {
				src: nextImage.imagePath,
				alt: nextImage.alt,
				css: {
					'position': 'absolute',
					'top': '0',
					'left': '0',
					'width': '100%',

				}
			});

			// Add new image to container
			$slideshowContainer.append($newImg);

			// Crossfade: fade in new image while fading out old
			$newImg.animate({ opacity: 1 }, 800);
			$slideshowImg.animate({ opacity: 0 }, 800, function() {
				// Remove old image and replace reference
				$(this).remove();
				$slideshowImg = $newImg;
			});
		}

		// Start the slideshow - change every 4 seconds
		if($('.stat-card.photo-card.slideshow').length > 0) {
