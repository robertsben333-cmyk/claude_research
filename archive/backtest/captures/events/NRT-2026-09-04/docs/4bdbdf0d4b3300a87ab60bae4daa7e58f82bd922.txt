         Why don’t NRT’s latest royalties fully reflect higher recent gas prices? 
          
       
       
        NRT’s royalty payments are based on amounts payable for the  prior quarter  under the Mobil and OEG Royalty Agreements. Because of this timing lag, royalties for the quarter ending  April 30, 2026  do not fully capture the recent period of higher gas prices.

async function regenFilingContent(event, filingId) {
  event.preventDefault();
  
  if (!confirm('This will regenerate all AI content for this filing. Continue?')) {
    return;
  }
  
  const button = event.target.closest('a');
  const originalTitle = button.title;
  button.style.pointerEvents = 'none';
  button.style.opacity = '0.5';
  button.title = 'Processing...';
  
  try {
    
    
    
    
    const cacheResponse = await fetch('/admin/sec-filings/clear-cache', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ filingId: filingId })
    });

    if (!cacheResponse.ok) {
      const cacheData = await cacheResponse.json().catch(() => ({}));
      throw new Error((cacheData.error && cacheData.error.message) || cacheData.error || cacheData.message || 'Failed to clear cache');
    }

    const processResponse = await fetch('/admin/sec-filings/process-filing', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ filingId: filingId })
    });

    if (!processResponse.ok) {
      throw new Error(`Process Full Filing failed: ${processResponse.status}`);
    }
    
    fetch(`https://api.stocktitan.net:10100/filings/generateExperts/${filingId}`, {
      headers: { 'x-admin-token': '' }
    })
      .catch(err => console.log('Expert generation request sent'));
    
    alert('AI content regeneration initiated successfully. The page will refresh in 5 seconds.');
    
    setTimeout(() => {
      window.location.reload();
    }, 5000);
    
  } catch (error) {
    alert('Error: ' + error.message);
    button.style.pointerEvents = '';
    button.style.opacity = '';
    button.title = originalTitle;
  }
}
