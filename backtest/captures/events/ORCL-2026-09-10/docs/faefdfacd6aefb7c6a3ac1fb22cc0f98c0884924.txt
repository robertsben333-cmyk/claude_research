  var debounceTimer = null;
  var controller = null;
  var activeIndex = -1;
  var currentItems = [];

  function escape(s) {
    return (s || '').replace(/[&<>"']/g, function(c) {
      return ({'&':'&amp;',' ':'&gt;','"':'&quot;',"'":'&#39;'})[c];
    });
  }

  function render(items, query) {
    var html = '';
    if (!items.length) {
      html += ' No matches for "' + escape(query) + '". ';
    } else {
      items.forEach(function(it) {
        var sym = escape(it.ticker || '');
        var name = escape(it.name || '');
        var href = it.url || ('/companies/' + (it.slug || ''));
        var logoUrl = 'https://images.financialmodelingprep.com/symbol/' + sym + '.png';
        html += ' '
              + ' '
              + ' '
              + ' '
              + ' '
              + ' ' + name + ' '
              + ' ' + sym + ' '
              + ' '
              + ' ';
      });
    }
    results.innerHTML = html;
    results.hidden = false;
    currentItems = Array.prototype.slice.call(results.querySelectorAll('.mlq-search-item'));
    activeIndex = -1;
  }

  function fetchResults(query) {
    if (controller) { try { controller.abort(); } catch (e) {} }
    controller = new AbortController();
    fetch('/companies/search/?q=' + encodeURIComponent(query), { signal: controller.signal })
      .then(function(r) { return r.ok ? r.json() : {}; })
      .then(function(data) { render((data && data.results) || [], query); })
      .catch(function() {});
  }

  input.addEventListener('input', function() {
    var q = input.value.trim();
    if (!q) { results.hidden = true; results.innerHTML = ''; currentItems = []; activeIndex = -1; return; }
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(function() { fetchResults(q); }, 150);
  });

  input.addEventListener('keydown', function(e) {
    if (results.hidden || !currentItems.length) { return; }
    if (e.key === 'ArrowDown') { e.preventDefault(); activeIndex = Math.min(activeIndex + 1, currentItems.length - 1); updateActive(); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); activeIndex = Math.max(activeIndex - 1, 0); updateActive(); }
    else if (e.key === 'Enter') { e.preventDefault(); var t = (activeIndex >= 0 ? currentItems[activeIndex] : currentItems[0]); if (t) window.location.href = t.getAttribute('data-href'); }
    else if (e.key === 'Escape') { results.hidden = true; input.blur(); }
  });

  function updateActive() {
    currentItems.forEach(function(el, i) { el.classList.toggle('is-active', i === activeIndex); if (i === activeIndex) el.scrollIntoView({ block: 'nearest' }); });
  }

  results.addEventListener('click', function(e) {
    var item = e.target.closest('.mlq-search-item, .mlq-search-ai');
    if (item) { e.preventDefault(); window.location.href = item.getAttribute('data-href'); }
  });

  document.addEventListener('click', function(e) {
    if (!input.contains(e.target) && !results.contains(e.target)) { results.hidden = true; }
  });
  input.addEventListener('focus', function() { if (currentItems.length) results.hidden = false; });
})();

       Oracle Reports $55.7B FY2026 Capex, Guides to $70B Net Outlay in FY2027 

      
       
         Jun 11, 2026 
         · 
         7:26 PM 
         · 
         by MLQ Agent 
         · 
         4 min read 
       

      
      
       
         Key points 
         
          
           Oracle spent $55.7B in capex during FY2026, up from $21.2B the prior year, and guided to $70B net project cash outlay for FY2027 — with gross capex of $90-$95B before customer prepayments   [1]    [2]   
          
           Cloud infrastructure (IaaS) revenue grew 93% YoY in Q4, reaching $5.8B for the quarter and $18.1B for the full year   [3]   
          
           Remaining performance obligations hit $638B, up 363% YoY, with $75B classified as prepaid or customer-supplied AI hardware   [3]   
          
           Oracle plans to raise approximately $40B in combined debt and equity in FY2027, including a previously announced $20B equity issuance   [2]   
          
           ORCL shares fell 9.3% to $182.64 on June 11, erasing roughly $55B in market capitalization despite the company beating Q4 revenue and EPS estimates   [4]   
          
         
       
      

      
      

      
      
       
         Oracle disclosed on June 10 that it spent $55.7 billion in capital expenditures during fiscal year 2026 — more than 2.6 times the $21.2 billion it spent the prior year — and guided to approximately $70 billion in net project cash outlay for FY2027 as it accelerates construction of AI-optimized data center infrastructure   [1]    [2]  . The gross capex figure for FY2027 is expected to land between $90 billion and $95 billion before accounting for $20-$25 billion in customer prepayments that offset Oracle's own capital requirements   [2]  .  The spending surge accompanied a strong top-line quarter: Q4 revenue hit $19.2 billion, up 21% year-over-year, beating the consensus estimate of $19.1 billion. Non-GAAP earnings per share came in at $2.11, ahead of the $1.96 forecast   [3]  . Cloud infrastructure revenue grew 93% in Q4, the company's fastest-growing segment and the engine behind the buildout   [3]  .  Wall Street focused on the cash burn. Oracle reported free cash flow of negative $23.7 billion for FY2026, driven by infrastructure investment that far outpaced the $32 billion in operating cash flow the company generated   [3]  . The company disclosed plans to raise approximately $40 billion in combined debt and equity during FY2027, including a previously announced $20 billion at-the-market equity offering, on top of the $43 billion in debt and $5 billion in equity already raised in FY2026   [2]  . Shares fell 9.3% to $182.64 on June 11   [4]  .

         The Numbers 
         
           Oracle's FY2026 capex of $55.7 billion exceeded analyst expectations of roughly $50.9 billion   [2]  . The company introduced a new reporting metric — 'project net cash outlay' — to distinguish between its gross capital expenditures and the portion offset by customer prepayments and timing effects. For FY2027, Oracle projects net cash outlay of approximately $70 billion, with gross capex of $90-$95 billion   [2]  .  Full-year revenue reached $67.4 billion, up 17% year-over-year. Cloud revenue totaled $34 billion, up 39%, with IaaS contributing $18.1 billion of that — a 77% increase   [3]  . The company confirmed its FY2027 revenue target of $90 billion and raised its non-GAAP EPS guidance to $8.05   [2]  .

         The Backlog 
         
           Remaining performance obligations — Oracle's measure of contracted but unrecognized revenue — reached $638 billion at the end of Q4, up 363% year-over-year and $85 billion higher than the prior quarter   [3]  . The figure includes approximately $75 billion classified as prepaid or customer-supplied AI hardware, which Oracle says reduces required capital for building AI data centers while supporting long-term revenue visibility   [3]  .  A significant portion of the RPO is tied to the Stargate joint venture with OpenAI and SoftBank, with reports attributing roughly $300 billion of the backlog to OpenAI contracts alone   [1]  . Oracle acknowledged concentration risk but emphasized that customer prepayments 'substantially reduce' required capital raises   [2]  .

         Data Center Capacity 
         
           Oracle added approximately 400 megawatts of data center capacity during Q4 FY2026 and expects to bring nearly 1 gigawatt online in the first quarter of FY2027   [1]  . The company has been expanding its cloud regions globally, building out GPU clusters optimized for large-scale AI training workloads.  CFO Hilary Maxson noted the company does not anticipate raising additional debt in calendar year 2026, suggesting the bulk of FY2027 financing will come through equity and cash flow   [1]  . CEO Clay Magouyrk acknowledged rising component costs — including memory, SSD, and hard drive prices — but stated Oracle has 'locked prices across the spectrum' to manage input costs   [1]  . 
         
       
      
      
       
         Financing Strategy 
         
           The scale of Oracle's financing requirements is notable even among hyperscalers. The company raised $43 billion in debt and $5 billion in equity during FY2026, and plans another $40 billion in combined financing for FY2027 — including the $20 billion at-the-market equity issuance already announced   [3]    [2]  . Oracle's total debt load has expanded significantly as it leverages its balance sheet to fund the AI buildout.  The deeply negative free cash flow of -$23.7 billion — despite $32 billion in operating cash flow — illustrates the gap between Oracle's cash generation and its investment pace   [3]  . The company is betting that its contracted backlog, now at $638 billion, will convert into revenue fast enough to justify the leverage. 
         
       
      
      
       
         Market Reaction 
         
           ORCL shares dropped 9.3% on June 11 to $182.64, wiping out roughly $55 billion in market capitalization   [4]  . The stock is now down 6.3% year-to-date and sits 47% below its 52-week high of $345.72   [4]  . The sell-off came despite Oracle beating estimates on both revenue and earnings — a signal that investors are focused on the trajectory of cash consumption rather than top-line growth.  The capex overshoot relative to estimates — $55.7 billion actual versus $50.9 billion expected — and the introduction of the $70 billion net outlay figure for FY2027 appear to have caught some analysts off guard   [2]  . Oracle now sits alongside Meta, Microsoft, Amazon, and Alphabet in the tier of companies spending tens of billions annually on AI infrastructure, a position it did not occupy two years ago.

  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var btn = form.querySelector('button');
    var input = form.querySelector('input[name=email]');
    btn.disabled = true;
    btn.textContent = 'Subscribing…';
    fetch('/newsletter/newsletters/signup/', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
        'X-Requested-With': 'XMLHttpRequest',
      },
      body: 'email=' + encodeURIComponent(input.value),
    })
      .then(function (r) { return r.json(); })
      .then(function (data) {
        if (data.status === 'success' || data.status === 'warning') {
          form.hidden = true;
          var success = form.parentElement.querySelector('.article-signup__success');
          if (success) {
            success.hidden = false;
            if (data.status === 'warning') success.textContent = "You're already subscribed!";
          }
        } else {
          btn.disabled = false;
          btn.textContent = 'Subscribe';
          alert(data.message || 'Something went wrong. Please try again.');
        }
      })
      .catch(function () {
        btn.disabled = false;
        btn.textContent = 'Subscribe';
      });
  });
});

    var offerId = overlay.getAttribute('data-offer-id');
    var offerUrl = overlay.getAttribute('data-offer-url');
    var ticker = overlay.getAttribute('data-ticker');

    var DISMISS_KEY = 'growth_modal_dismissed';
    var SIGNUP_KEY = 'growth_modal_signed_up';

    // Don't show if dismissed in last 3 days or signed up in last 30
    var dismissed = localStorage.getItem(DISMISS_KEY);
    if (dismissed && (Date.now() - parseInt(dismissed))   0.5) {
            showModal();
            window.removeEventListener('scroll', checkScroll);
        }
    }
    window.addEventListener('scroll', checkScroll);

    function closeModal() {
        overlay.classList.remove('visible');
        localStorage.setItem(DISMISS_KEY, Date.now().toString());
        clearTimeout(timer);
        window.removeEventListener('scroll', checkScroll);
    }

    closeBtn.addEventListener('click', closeModal);
    overlay.addEventListener('click', function(e) {
        if (e.target === overlay) closeModal();
    });

    btn.addEventListener('click', function() {
        var email = input.value.trim();
        if (!email || !email.includes('@')) {
            input.style.borderColor = '#ef4444';
            return;
        }
        btn.textContent = '...';
        btn.disabled = true;

        fetch('/growth/capture/', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({
                email: email,
                offer_id: offerId || null,
                ticker: ticker || null,
                source_page: window.location.pathname,
            })
        })
        .then(function(r) { return r.json(); })
        .then(function(data) {
            if (data.success) {
                localStorage.setItem(SIGNUP_KEY, Date.now().toString());

                    content.innerHTML = '  Your report is ready   Read it now  or check your inbox.  ';

                }
                setTimeout(function() { overlay.classList.remove('visible'); }, 4000);
            } else {
                btn.textContent = 'Try again';
                btn.disabled = false;
            }
        })
        .catch(function() {
            btn.textContent = 'Try again';
            btn.disabled = false;
        });
    });

    input.addEventListener('keypress', function(e) {
        if (e.key === 'Enter') btn.click();
    });
