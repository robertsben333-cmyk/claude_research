    function mapTierToName(tier){
      var tierNumber = parseInt(tier, 10);
      if (isNaN(tierNumber) || tierNumber  = 4) return 'enterprise';
      return 'free';
    }


        $('[data-toggle="tooltip"]').tooltip()
        //Search
        function debounce(func, wait, immediate) {
            var timeout;
            return function() {
                var context = this, args = arguments;
                var later = function() {
                    timeout = null;
                    if (!immediate) func.apply(context, args);
                };
                var callNow = immediate && !timeout;
                clearTimeout(timeout);
                timeout = setTimeout(later, wait);
                if (callNow) func.apply(context, args);
            };
        }

        // AJAX search function
        var executeSearch = debounce(function() {
            var inputVal = $('#search_companies').val();
            $("#search_results").addClass("preview-list show");
            if (inputVal.length) {
                $("#search_results").html(" ");
                $.get("https://altindex.com/php/ajax/search_companies.php", {
                    term: inputVal
                }).done(function(data) {
                    $("#search_results").html(data);
                });
            } else {
                $("#search_results").removeClass("preview-list show");
            }
        }, 250); // Adjust the delay as needed

        // Bind the input event to the debounced function
        $('#search_companies').on('input', function(event) {
            executeSearch();
            event.stopPropagation();
        });
        // Set search input value on click of result item
        $(document).on("click", ".result p", function(){
            $(this).parents(".search-box").find('input[type="text"]').val($(this).text());
            $(this).parent(".result").empty();
        });

        //change password


        $('.search_highlight').click(function () { 
          $('#search_form').contents().appendTo('#search_companies_modul');
        });
        $('#searchModal').on('hidden.bs.modal', function () { 
          window.location.reload();
        });
