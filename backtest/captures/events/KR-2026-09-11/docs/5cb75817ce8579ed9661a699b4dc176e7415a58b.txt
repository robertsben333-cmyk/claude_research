  async function logoutUser() {

     Runs independently of the shortcuts card below.
  ===================================================================== */
    const initFormSwitcher = () => {
      const forms = [...document.querySelectorAll('.shortcuts-form')];

      // Remember each form's natural display so we can restore it when shown
      // (works whether or not the wrappers are set to Display:None in Webflow).
      forms.forEach((form) => {
        const d = getComputedStyle(form).display;
        form.__display = d && d !== 'none' ? d : 'block';
      });

      const showForm = (key) => {
        forms.forEach((form) => {
          form.style.display = form.getAttribute('data-form') === key ? form.__display : 'none';
        });
      };

      // Each option: sync the label, reveal its form, close the dropdown.
      const dropdownOptions = [...document.querySelectorAll('.placer-100-dropdown-option')];
      dropdownOptions.forEach((option) => {
        option.addEventListener('click', () => {
          const currentOptionText = document.querySelector('.dropdown-placeholder').textContent;
          const selectedText = option.textContent;

          if (selectedText !== currentOptionText) {
            document.querySelectorAll('.dropdown-placeholder').forEach((el) => {
              el.textContent = selectedText;
            });
          }

          showForm(option.getAttribute('data-form'));
          $('.placer-100-dropdown').trigger('w-close');
        });
      });

      // Show the default form on load (matches the first option in the DOM).
      const defaultOption = document.querySelector('.placer-100-dropdown-option');
      if (defaultOption) showForm(defaultOption.getAttribute('data-form'));
    };

    /* =====================================================================
     SECTION 2 — Shortcuts widget (floating card: open/close + inner views)
  ===================================================================== */
    const initShortcutsCard = () => {
      const WRAPPER = '.shortcuts-wrapper';
      const BUTTON = '.shortcuts-button';
      const MENU = '.shortcuts-menu'; // the card — opens/closes
      const HOME = '.shortcuts-home'; // default inner view
      const PANEL = '.shortcuts-panel'; // the other inner views
      const ITEM = '.shortcut-menu-item';
      const BACK = '.shortcuts-back';
      const CLOSE = '.shortcuts-close-btn';
      const CARD_MS = 200; // card open/close fade
      const SWAP_MS = 140; // inner content swap fade

      const wrapper = document.querySelector(WRAPPER);
      if (!wrapper) return;

      const button = wrapper.querySelector(BUTTON);
      const menu = wrapper.querySelector(MENU);
      const home = wrapper.querySelector(HOME);
      if (!button || !menu || !home) return;

      const panels = [...wrapper.querySelectorAll(PANEL)];
      const innerViews = [home, ...panels];
      let isOpen = false; // is the card open?
      let currentInner = home; // which inner view is showing
      let hideTimer;

      // ---- prep the card ----
      const cardDisplay = getComputedStyle(menu).display;
      menu.__open = cardDisplay && cardDisplay !== 'none' ? cardDisplay : 'flex';
      menu.style.transition = `opacity ${CARD_MS}ms ease, transform ${CARD_MS}ms ease`;
      menu.style.display = 'none';
      menu.setAttribute('aria-hidden', 'true');

      // ---- prep inner views ----
      innerViews.forEach((view) => {
        const d = getComputedStyle(view).display;
        view.__open = d && d !== 'none' ? d : 'flex';
        view.style.transition = `opacity ${SWAP_MS}ms ease, transform ${SWAP_MS}ms ease`;
      });

      // Swap which inner view is visible (card unaffected).
      const setInner = (view, animate) => {
        innerViews.forEach((v) => {
          if (v !== view) {
            v.style.display = 'none';
            v.setAttribute('aria-hidden', 'true');
          }
        });

        view.style.display = view.__open;
        view.setAttribute('aria-hidden', 'false');

        if (animate) {
          view.style.opacity = '0';
          view.style.transform = 'translateY(6px)';
          requestAnimationFrame(() => {
            view.style.opacity = '1';
            view.style.transform = 'translateY(0)';
          });
        } else {
          view.style.opacity = '1';
          view.style.transform = 'translateY(0)';
        }

        currentInner = view;
      };

      const openMenu = () => {
        if (isOpen) return;
        isOpen = true;
        clearTimeout(hideTimer);

        setInner(home, false); // always reopen at Home
        button.style.display = 'none'; // hide the pill
        menu.style.display = menu.__open;
        menu.style.opacity = '0';
        menu.style.transform = 'translateY(8px)';
        menu.setAttribute('aria-hidden', 'false');

        requestAnimationFrame(() => {
          menu.style.opacity = '1';
          menu.style.transform = 'translateY(0)';
        });
      };

      const closeMenu = () => {
        if (!isOpen) return;
        isOpen = false;

        menu.style.opacity = '0';
        menu.style.transform = 'translateY(8px)';
        menu.setAttribute('aria-hidden', 'true');

        hideTimer = setTimeout(() => {
          menu.style.display = 'none';
          button.style.display = ''; // pill returns
          setInner(home, false); // reset for next open
        }, CARD_MS);
      };

      const panelFor = (key) => panels.find((p) => p.getAttribute('data-panel') === key) || null;

      // ---- initial state ----
      setInner(home, false); // start on Home

      // Pill -> open the card (at Home).
      button.addEventListener('click', (e) => {
        e.preventDefault();
        openMenu();
      });

      // Home menu item -> swap inner content to its panel (card stays open).
      [...home.querySelectorAll(ITEM)].forEach((item) => {
        item.addEventListener('click', (e) => {
          const key = item.getAttribute('data-shortcut');
          const panel = key ? panelFor(key) : null;
          if (!panel) return; // no destination -> leave the link alone
          e.preventDefault();
          setInner(panel, true);
        });
      });

      // Back -> Home ; Close -> close the whole card (delegated).
      wrapper.addEventListener('click', (e) => {
        if (e.target.closest(BACK)) {
          e.preventDefault();
          setInner(home, true);
          return;
        }
        if (e.target.closest(CLOSE)) {
          e.preventDefault();
          closeMenu();
          return;
        }
      });

      // Outside click -> close. Skipped while a data-keep-open view (form) is showing.
      document.addEventListener('click', (e) => {
        if (!isOpen) return;
        if (wrapper.contains(e.target)) return;
        if (currentInner && currentInner.hasAttribute('data-keep-open')) return;
        closeMenu();
      });

      // Esc -> close.
      document.addEventListener('keydown', (e) => {
        if (isOpen && (e.key === 'Escape' || e.key === 'Esc')) closeMenu();
      });

      // ---- Optional in-panel tabs (Pricing / Demo, etc.) ----
      panels.forEach((panel) => {
        const btns = [...panel.querySelectorAll('[data-tab]')];
        if (!btns.length) return;

        const contents = [...panel.querySelectorAll('[data-tab-content]')];

        const activate = (key) => {
          btns.forEach((b) => {
            b.classList.toggle('is-active', b.getAttribute('data-tab') === key);
          });
          contents.forEach((c) => {
            c.style.display = c.getAttribute('data-tab-content') === key ? '' : 'none';
          });
        };

        btns.forEach((b) => {
          b.addEventListener('click', (e) => {
            e.preventDefault();
            activate(b.getAttribute('data-tab'));
          });
        });

        activate(btns[0].getAttribute('data-tab'));
      });
    };

    /* ---- boot both ---- */
    const init = () => {
      initFormSwitcher();
      initShortcutsCard();
    };

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', init);
    } else {
      init();
    }
  })();

  (function () {
    var DEBUG = false;

    var securedCount = 0;
    var skippedCount = 0;

    function log() {
      if (DEBUG) console.log.apply(console, ['[link-secure]'].concat([].slice.call(arguments)));
    }

    function secureLink(a) {
      try {
        if (!a || a.tagName !== 'A' || a.target !== '_blank') return;

        var href = a.getAttribute('href') || '';
        if (/^(mailto:|tel:|sms:|javascript:|data:)/i.test(href)) {
          skippedCount++;
          log('skipped non-web protocol', href, a);
          return;
        }

        var beforeRel = (a.getAttribute('rel') || '').trim();
        var rel = beforeRel ? beforeRel.split(/\s+/) : [];
        var added = [];

        if (rel.indexOf('noopener') === -1) { rel.push('noopener'); added.push('noopener'); }
        if (rel.indexOf('noreferrer') === -1) { rel.push('noreferrer'); added.push('noreferrer'); }

        if (added.length) {
          a.setAttribute('rel', rel.join(' '));
          securedCount++;
          log('secured', { href: href, added: added, afterRel: rel.join(' ') }, a);
        }
      } catch (err) {
        if (DEBUG) console.warn('[link-secure] secureLink error', err, a);
      }
    }

    // Existing + future links
    PLACER.each('a[target="_blank"]', secureLink);

    // A link can also become _blank by attribute change
    PLACER.onMutation(function (m) {
      if (m.type === 'attributes' && m.attributeName === 'target' && m.target.tagName === 'A') {
        secureLink(m.target);
      }
    });

    if (DEBUG) {
      setInterval(function () {
        console.log('[link-secure] secured: ' + securedCount + ', skipped: ' + skippedCount);
      }, 10000);
    }
  })();

  (function () {
    var FOCUSABLE_SELECTOR = [
      'a[href]',
      'button:not([disabled])',
      'textarea:not([disabled])',
      'input:not([disabled])',
      'select:not([disabled])',
      '[tabindex]:not([tabindex="-1"])'
    ].join(',');

    var activeModal = null;
    var lastTrigger = null;

    function getFocusableElements(container) {
      if (!container) return [];
      return Array.from(container.querySelectorAll(FOCUSABLE_SELECTOR)).filter(function (el) {
        return !!(el.offsetWidth || el.offsetHeight || el.getClientRects().length);
      });
    }

    function setModalState(modal, isOpen) {
      if (!modal) return;

      if (isOpen) {
        modal.classList.add('is-open', 'show');
        modal.style.display = 'flex';
        modal.setAttribute('aria-hidden', 'false');
        modal.setAttribute('aria-modal', 'true');
        if (!modal.hasAttribute('role')) modal.setAttribute('role', 'dialog');
      } else {
        modal.classList.remove('is-open', 'show');
        modal.style.display = 'none';
        modal.setAttribute('aria-hidden', 'true');
      }
    }

    function focusFirstInModal() {
      if (!activeModal) return;

      // WCAG 2.4.3 - land on the Register button first when present
      var signupBtn = activeModal.querySelector('#signup-btn');
      if (signupBtn) {
        signupBtn.focus();
        return;
      }

      var focusables = getFocusableElements(activeModal);
      if (focusables.length) {
        focusables[0].focus();
      } else {
        activeModal.setAttribute('tabindex', '-1');
        activeModal.focus();
      }
    }

    function openModal(modal, trigger) {
      if (!modal) return;
      activeModal = modal;
      lastTrigger = trigger || null;
      setModalState(modal, true);
      setTimeout(focusFirstInModal, 50); // wait for render
    }

    function closeModal(modal) {
      if (!modal) return;
      setModalState(modal, false);
      if (activeModal === modal) activeModal = null;
      if (lastTrigger && lastTrigger.focus) lastTrigger.focus();
      lastTrigger = null;
    }

    window.focusFirstInOpenModal = focusFirstInModal;

    // Single capture-phase delegate for both open and close
    window.addEventListener('click', function (e) {
      var opener = e.target.closest('.js-modal-open');
      if (opener) {
        var selector = opener.getAttribute('data-modal-target');
        if (!selector) return;
        var modal = document.querySelector(selector);
        if (!modal) return;
        e.preventDefault();
        openModal(modal, opener);
        return;
      }

      var closer = e.target.closest('.js-modal-close');
      if (!closer) return;

      var owner = closer.closest('.js-modal');
      if (!owner) {
        if (activeModal) closeModal(activeModal);
        return;
      }
      e.preventDefault();
      closeModal(owner);
    }, true);

    document.addEventListener('keydown', function (e) {
      if (!activeModal) return;

      if (e.key === 'Escape') {
        e.preventDefault();
        closeModal(activeModal);
        return;
      }

      if (e.key !== 'Tab') return;

      var focusables = getFocusableElements(activeModal);
      if (!focusables.length) {
        e.preventDefault();
        return;
      }

      var first = focusables[0];
      var last = focusables[focusables.length - 1];
      var active = document.activeElement;

      if (!activeModal.contains(active)) {
        e.preventDefault();
        first.focus();
        return;
      }

      if (e.shiftKey && active === first) {
        e.preventDefault();
        last.focus();
      } else if (!e.shiftKey && active === last) {
        e.preventDefault();
        first.focus();
      }
    });
  })();

    var mainContent = document.getElementById('main-content');

  (function () {
    var TRUSTED_ORIGINS = [
      'https://embedded-widget.placer.ai',
      'https://analytics.placer.ai'
    ];

    var identifiedEmail = null;
    var polling = false;

    function heapIdentify(email) {
      // window.heap may be undefined - Heap is C0002-gated, so a bare `heap`
      // reference throws a ReferenceError when consent was declined.
      if (window.heap && typeof window.heap.identify === 'function') {
        window.heap.identify(email);
      }
    }

    function hsIdentify(email) {
      window._hsq = window._hsq || [];
      _hsq.push(['identify', { email: email }]);
      _hsq.push(['trackPageView']); // identify only attaches to the next event
    }

    function identify(email) {
      if (!email || email === identifiedEmail) return; // dedupe per page load
      identifiedEmail = email;
      heapIdentify(email);
      hsIdentify(email);
    }

    // --- Path 1: already logged in - poll ew.js user store if present
    function pollPlacerUser(timeoutMs, intervalMs) {
      if (polling) return;
      polling = true;

      timeoutMs = timeoutMs || 10000;
      intervalMs = intervalMs || 250;

      var start = Date.now();
      var timer = setInterval(function () {
        var store = window.$app &&
                    window.$app.components &&
                    window.$app.components.placer_user &&
                    window.$app.components.placer_user.store;

        if (store && store.isAuthenticated && store.email) {
          clearInterval(timer);
          polling = false;
          identify(store.email);
        } else if (Date.now() - start > timeoutMs) {
          clearInterval(timer); // no ew.js or not logged in - silent no-op
          polling = false;
        }
      }, intervalMs);
    }

    PLACER.ready(function () { pollPlacerUser(); });

    // --- Path 2: in-page login / sign-up via the portable auth iframe
    // --- Path 3: HubSpot form submissions relayed by postMessage
    window.addEventListener('message', function (e) {
      var d = e.data;
      if (!d) return;

      // Path 2 requires a trusted origin - it carries user PII directly
      if (TRUSTED_ORIGINS.indexOf(e.origin) !== -1) {
        if (d.type === 'PLACER_USER' && d.user && d.user.email) {
          identify(d.user.email);
          return;
        }
        if (d === 'LOGIN_SUCCESSES') {
          pollPlacerUser(); // no payload - re-poll the store for the fresh email
          return;
        }
      }

      // Path 3: iframe-embedded HS forms on another subdomain never stitch the
      // hubspotutk cookie, so identify from the submitted values instead.
      if (d.type === 'hsFormCallback' && d.eventName === 'onFormSubmitted') {
        var values = (d.data && d.data.submissionValues) || {};
        var email = PLACER.findEmail(values);
        if (email) identify(email);
      }
    });
  })();
