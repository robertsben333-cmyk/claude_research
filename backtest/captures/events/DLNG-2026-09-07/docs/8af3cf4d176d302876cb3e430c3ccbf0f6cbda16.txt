    function catapultSetCookie(cookieName, cookieValue, nDays) {
        var today = new Date();
        var expire = new Date();
        expire.setTime(today.getTime() + 3600000*24*nDays);
        document.cookie = cookieName+"="+escape(cookieValue)+ ";expires=" + expire.toGMTString() +"; path=/";
    }
    function catapultAcceptCookies() {
        catapultSetCookie('catAccCookies', 1, 1);
        document.querySelector('.cookie-bar').classList.add('cookie-bar-hidden');
    }

    function stickySkinAds(){ 
        if(($('#sideAd').length > 0) && ($(window).width() > 768)){
            $(window).scroll(function(){
                var topHeight = 30;
                if($('header').hasClass('is-pinned')){
                    topHeight += $("header").outerHeight();
                    $('#sideAd').addClass('sideAd-sticky').css('top', topHeight + 'px');
                    $('.skyscraper-right').css('left', '1220px').css('right',  'unset');
                }else{
                    topHeight += $("header").outerHeight() + $(".header-ad").outerHeight();
                    $('#sideAd').removeClass('sideAd-sticky').css('top', '0px');
                    $('.skyscraper-right').css('right', '-335px').css('left',  'unset');
                }
            });
        }
    }
