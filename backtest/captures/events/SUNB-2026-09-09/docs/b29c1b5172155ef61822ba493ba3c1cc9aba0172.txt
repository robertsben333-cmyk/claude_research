                Company Overview 
                
                     Sunbelt Rentals Holdings, Inc. is a major equipment rental company that provides a broad range of rental solutions to construction, industrial, municipal and specialty markets. The company's business centers on short- and long-term rental of equipment used for earthmoving, aerial access, material handling, power generation, climate control, pumps and general tools, supporting customers who need flexible access to capital-intensive assets without the expense of ownership. 

 In addition to physical equipment, Sunbelt Rentals offers complementary services designed to keep projects running efficiently. These include on-site delivery and pickup, maintenance and repair services, fleet management and logistics, safety training and technical support. The company markets specialty rental packages and turnkey solutions for sectors such as infrastructure, oil and gas, manufacturing, events and emergency response, enabling clients to scale resources quickly for peak workloads or one-off projects. 

 Sunbelt Rentals operates a network of branches and distribution centers across North America, serving contractors, building owners, industrial operators and public agencies. Its operating model emphasizes local availability, rapid service response and asset utilization across a decentralized footprint. The company positions itself as a solutions partner that combines inventory breadth with operational support to reduce downtime and help customers manage project schedules and costs more predictably.         function SendMagicLinkOnModal(token) {
            var notice = document.getElementById('loginOnModalWarning');
            var email = document.getElementById('txtLoginOnModalEmail').value;
            if (email == "") {
                notice.innerHTML = " Please enter your email address. ";
                return;
            }
            document.getElementById('btnMagicLinkOnModal').classList.add("active-loading");
            var returnUrl = window.location.pathname + window.location.search;
            console.log('[magiclink] sending', { tokenLength: (token || '').length, returnUrlLength: returnUrl.length });

            // POST, not GET: the reCAPTCHA token alone can blow past the 2048-char query string limit
            $.ajax({
                type: 'POST',
                url: '/scripts/modal/EmailChecker.ashx',
                data: {
                    command: 'magiclink',
                    email: email,
                    returnUrl: returnUrl,
                    recaptchaToken: token
                },
                success: function (data) {
                    var btn = document.getElementById('btnMagicLinkOnModal');
                    btn.classList.remove("active-loading");
                    console.log('[magiclink] response', data);
                    if (data.indexOf('c-red') > -1) {
                        notice.innerHTML = data;
                        // v2 invisible tokens are single-use â€” reset or the next click yields nothing
                        if (typeof grecaptcha !== 'undefined') { grecaptcha.reset(); }
                    } else {
                        notice.innerHTML = " " + data + " ";
                        btn.disabled = true;
                        btn.classList.remove('btn-secondary');
                        btn.classList.add('btn-disabled');
                    }
                },
                error: function (xhr, textStatus, errorThrown) {
                    document.getElementById('btnMagicLinkOnModal').classList.remove("active-loading");
                    console.error('[magiclink] failed', {
                        status: xhr.status,               // 404 = IIS rejected the request (404.15 = query string too long)
                        statusText: xhr.statusText,
                        textStatus: textStatus,
                        errorThrown: errorThrown,
                        tokenLength: (token || '').length,
                        response: (xhr.responseText || '').substring(0, 1000)
                    });
                    notice.innerHTML = " Something went wrong sending your login link. Please try again. ";
                    if (typeof grecaptcha !== 'undefined') { grecaptcha.reset(); }
                }
            });
        }

        // Toggle modal magic link button style based on email input
        document.addEventListener('input', function (e) {
            if (e.target.id === 'txtLoginOnModalEmail') {
                var btn = document.getElementById('btnMagicLinkOnModal');
                if (btn) {
                    var hasEmail = e.target.value.trim() !== '';
                    btn.classList.toggle('btn-disabled', !hasEmail);
                    btn.classList.toggle('btn-secondary', hasEmail);
                }
            }
        });
