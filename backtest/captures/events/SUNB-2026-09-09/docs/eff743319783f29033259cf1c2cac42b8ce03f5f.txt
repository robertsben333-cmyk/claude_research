                        let debounceSearchTimer;
                        let lastSearchRequest = null;

                        $(document).on("keyup keydown", "#searchParam", function(e) {
                            const $input = $(this);
                            const keyCode = e.keyCode || e.which;

                            if (keyCode === 13) {
                                e.preventDefault();
                                window.location.href = '/search?param=' + $input.val().trim();
                                return false;
                            }

                            const value = $input.val().trim();

                            if (value === "") {
                                $("#search-result").html('').hide();
                                return;
                            }

                            // Clear previous debounce timer
                            clearTimeout(debounceSearchTimer);

                            // Debounce execution by 500ms
                            debounceSearchTimer = setTimeout(() => {
                                removeLoad($input);
                                $input.after(loaderSpinner);

                                // Abort the previous request if it's still pending
                                if (lastSearchRequest && lastSearchRequest.readyState !== 4) {
                                    lastSearchRequest.abort();
                                }


                                // New AJAX request
                                lastSearchRequest = $.get("https://kalkinemedia.com/search-suggestions", {
                                    query: value
                                }, function(data) {
                                    if (value === $("#searchParam").val()) {
                                        $("#search-result").html(data).show();
                                    }
                                    removeLoad($input);
                                });
                            }, 500);
                        });

                        function removeLoad(val) {
                            val.parent().find("div.spinner-border").remove();
                        }

                        $(document).on("click", function (e) {
                            if (
                                !$(e.target).closest("#search-result").length
                            ) {
                                $("#search-result").hide();
                                $("#searchParam").val('');
                            }

        var ARTICLE_BLOCKER_DELAY_MS = 10000;
        var blocker = document.getElementById('mandatoryLead');
        var blockerTimer = null;
        var blockerIsActive = false;

            return;

        function getFocusableElements() {
            return Array.prototype.slice.call(blocker.querySelectorAll(
                'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
            )).filter(function (element) {
                return element.offsetParent !== null;
            });
        }

        function keepBlockerLocked(event) {
            if (!blockerIsActive) {
                return;
            }

            if (event.key === 'Escape') {
                event.preventDefault();
                event.stopImmediatePropagation();
                return;
            }

            if (event.key !== 'Tab') {
                return;
            }

            var focusableElements = getFocusableElements();
            if (!focusableElements.length) {
                event.preventDefault();
                blocker.focus();
                return;
            }

            var firstElement = focusableElements[0];
            var lastElement = focusableElements[focusableElements.length - 1];

            if (event.shiftKey && document.activeElement === firstElement) {
                event.preventDefault();
                lastElement.focus();
            } else if (!event.shiftKey && document.activeElement === lastElement) {
                event.preventDefault();
                firstElement.focus();
            }
        }

        function scrollPageBehindBlocker(event) {
            if (!blockerIsActive || event.ctrlKey) {
                return;
            }

            var dialog = event.target.closest('.article-blocker__dialog');

            if (dialog && dialog.scrollHeight > dialog.clientHeight) {
                var scrollingUp = event.deltaY  = dialog.scrollHeight;

                if ((!scrollingUp && !atBottom) || (scrollingUp && !atTop)) {
                    return;
                }
            }

            event.preventDefault();
            window.scrollBy({
                top: event.deltaY,
                left: 0,
                behavior: 'auto'
            });
        }

        function showArticleBlocker() {
            blockerIsActive = true;
            blocker.hidden = false;
            blocker.setAttribute('tabindex', '-1');
            document.addEventListener('keydown', keepBlockerLocked, true);

            var focusableElements = getFocusableElements();
            (focusableElements[0] || blocker).focus();
        }

        function resetArticleBlockerTimer() {
            window.clearTimeout(blockerTimer);
            blockerIsActive = false;
            blocker.hidden = true;
            document.removeEventListener('keydown', keepBlockerLocked, true);
            blockerTimer = window.setTimeout(showArticleBlocker, ARTICLE_BLOCKER_DELAY_MS);
        }

        resetArticleBlockerTimer();
        blocker.addEventListener('wheel', scrollPageBehindBlocker, { passive: false });

        window.addEventListener('pagehide', function () {
            window.clearTimeout(blockerTimer);
        });

        window.addEventListener('pageshow', function (event) {
            if (event.persisted) {
                resetArticleBlockerTimer();
            }
        });

            function cookieAccept() {
                $("#cookieConsentChanged").css("display", "none");
                var acceptDate = new Date;
                acceptDate.setDate(acceptDate.getDate() + 7), document.cookie = "CookieAccept=Accept; true; expires=" +
                    acceptDate;
            }

            function cookieDecline() {
                $("#cookieConsentChanged").css("display", "none");
                var declieDate = new Date;
                declieDate.setDate(declieDate.getDate() + 30), document.cookie = "CookieDecline=Decline; true; expires=" +
                    declieDate;
            }

            function getCookie(cookie_name) {
                if (document.cookie.length > 0) {
                    cookie_start = document.cookie.indexOf(cookie_name + "=");
                    if (cookie_start != -1) {
                        cookie_start = cookie_start + cookie_name.length + 1;
                        cookie_end = document.cookie.indexOf(";", cookie_start);
                        if (cookie_end == -1) {
                            cookie_end = document.cookie.length;
                        }
                        return unescape(document.cookie.substring(cookie_start, cookie_end));
                    }
                }
                return "";
            }
