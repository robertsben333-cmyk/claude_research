                Company Overview 
                
                     Cooper Companies, Inc. (NASDAQ: COO) is a global medical device company headquartered in San Ramon, California. Founded in 1958, the company has grown through strategic acquisitions and organic development to become a major provider of vision care and women’s health products. Cooper Companies operates through two primary business segments—CooperVision and CooperSurgical—each serving specialized markets within the healthcare industry. 

 The CooperVision segment develops, manufactures and markets a broad range of soft contact lenses, as well as related accessories. Its portfolio includes single‐use, multifocal, toric and specialty lenses for presbyopia and astigmatism, supported by proprietary materials and manufacturing technologies. CooperSurgical focuses on women’s health and fertility solutions, offering products such as surgical instruments, diagnostic devices, and assisted reproductive technology tools designed for obstetrics, gynecology and fertility clinics. 

 With a presence in over 100 countries, CooperCompanies serves markets across North America, Europe, Asia Pacific, Latin America and the Middle East & Africa. The company maintains manufacturing facilities and R&D centers in key regions, enabling close collaboration with healthcare professionals and customers to address evolving medical and patient needs. 

 CooperCompanies is guided by an experienced leadership team and board of directors committed to innovation, quality and compliance. Through ongoing investment in research and development, the company aims to expand its product offerings, enhance clinical outcomes and support the global healthcare community.    AI Generated.         function SendMagicLinkOnModal(token) {
            var notice = document.getElementById('loginOnModalWarning');
            var email = document.getElementById('txtLoginOnModalEmail').value;
            if (email == "") {
                notice.innerHTML = " Please enter your email address. ";
                return;
            }
            document.getElementById('btnMagicLinkOnModal').classList.add("active-loading");
            var returnUrl = window.location.pathname + window.location.search;
            console.log('[magiclink] sending', { tokenLength: (token || '').length, returnUrlLength: returnUrl.length });

            // POST, not GET: the reCAPTCHA token alone can blow past the 2048-char query string limit
            $.ajax({
                type: 'POST',
                url: '/scripts/modal/EmailChecker.ashx',
                data: {
                    command: 'magiclink',
                    email: email,
                    returnUrl: returnUrl,
                    recaptchaToken: token
                },
                success: function (data) {
                    var btn = document.getElementById('btnMagicLinkOnModal');
                    btn.classList.remove("active-loading");
                    console.log('[magiclink] response', data);
                    if (data.indexOf('c-red') > -1) {
                        notice.innerHTML = data;
                        // v2 invisible tokens are single-use â€” reset or the next click yields nothing
                        if (typeof grecaptcha !== 'undefined') { grecaptcha.reset(); }
                    } else {
                        notice.innerHTML = " " + data + " ";
                        btn.disabled = true;
                        btn.classList.remove('btn-secondary');
                        btn.classList.add('btn-disabled');
                    }
                },
                error: function (xhr, textStatus, errorThrown) {
                    document.getElementById('btnMagicLinkOnModal').classList.remove("active-loading");
                    console.error('[magiclink] failed', {
                        status: xhr.status,               // 404 = IIS rejected the request (404.15 = query string too long)
                        statusText: xhr.statusText,
                        textStatus: textStatus,
                        errorThrown: errorThrown,
                        tokenLength: (token || '').length,
                        response: (xhr.responseText || '').substring(0, 1000)
                    });
                    notice.innerHTML = " Something went wrong sending your login link. Please try again. ";
                    if (typeof grecaptcha !== 'undefined') { grecaptcha.reset(); }
                }
            });
        }

        // Toggle modal magic link button style based on email input

                    btn.classList.toggle('btn-secondary', hasEmail);
                }
            }
        });
